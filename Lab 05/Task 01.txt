<h3>Enter numbers one by one. Enter 0 to stop and see the sum.</h3>
<p id="demo"></p>

<script>
var num;
var sum = 0;

while (true) {
  num = prompt("Enter a number (enter 0 to end):", 0);
  num = Number(num);

  if (num == 0) {
    break;
  }

  sum = sum + num;
}

document.getElementById("demo").innerHTML = "The sum is: " + sum;
</script>
