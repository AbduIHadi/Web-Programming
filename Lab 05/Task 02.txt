<script>
function askQuestion() {
  var correctAnswer = "Praia";
  var answer = prompt("What is the capital of Cabo Verde?", "");

  if (answer == correctAnswer) {
    document.getElementById("demo").innerHTML = "Success! That is correct.";
  } else {
    document.getElementById("demo").innerHTML = "Error! That is not correct.";
  }
}
