<h3>Enter a message and how many times to repeat it</h3>
<p id="demo"></p>

<script>
var count = prompt("How many times should the message repeat?", 0);
count = Number(count);

var message = prompt("Enter your message:", "");

var output = "";
for (var i = 1; i <= count; i++) {
  output = output + message + "<br>";
}

document.getElementById("demo").innerHTML = output;
</script>
